Q · query

Az aktuális pozíció vektora, amellyel illeszkedést keresünk.

Q = X WQT × dₖ

K · key

Minden pozíció kulcsvektora, amelyhez a queryt mérjük.

K = X WKT × dₖ

V · value

A vektor, amely a keverékbe kerül, ha az adott pozíció súlyt kap.

V = X WVT × dᵥ

"""A diasor kis GPT-je. Futtatás: python3 mini_gpt.py"""
from math import exp, log

TOKENS = ("A", " macska", " ül", " alszik", "<EOS>")
EMBEDDINGS = ((0, 1), (0.75, 0), (0, 1), (0, 0), (0, 0))
POSITIONS = ((0, 0), (0.25, 0), (0, 0))
W_UL = 0.4 * log(4)


def softmax(scores):
    peak = max(scores)
    exp_scores = [exp(s - peak) for s in scores]
    return [s / sum(exp_scores) for s in exp_scores]


def forward(ids, w_ul=W_UL, w_value_x=2, w_key=log(3)):
    x = [
        tuple(e + p for e, p in zip(EMBEDDINGS[t], POSITIONS[i]))
        for i, t in enumerate(ids)
    ]
    q = [a for a, b in x]
    k = [w_key * a for a, b in x]
    v = [(w_value_x * a, 4 * b) for a, b in x]

    positions = []
    for i in range(len(x)):
        # Az i. sorban csak az i-ig terjedő pontszámokat vesszük figyelembe.
        scores = [q[i] * k[j] for j in range(i + 1)]
        weights = softmax(scores)
        attention = tuple(
            sum(w * v[j][d] for j, w in enumerate(weights)) for d in range(2)
        )
        y = tuple(a + b for a, b in zip(x[i], attention))
        g = max(0, y[0] - 2 * y[1])
        h = (y[0], y[1] + 2 * g)
        logits = (0, 0, w_ul * h[0], 0.4 * log(2) * h[0], log(2) * (h[1] - 2))
        positions.append({
            "x": x[i], "q": q[i], "k": k[i], "v": v[i],
            "scores": scores, "weights": weights, "attention": attention,
            "y": y, "g": g, "h": h, "logits": logits, "p": softmax(logits),
        })
    return positions


def update(w_ul=W_UL, learning_rate=0.1):
    row = forward([0, 1], w_ul=w_ul)[1]
    grad = (row["p"][2] - 1) * row["h"][0]
    return w_ul - learning_rate * grad


if __name__ == "__main__":
    targets = [1, 2, 4]
    rows = forward([0, 1, 2])
    for i, (row, target) in enumerate(zip(rows, targets), 1):
        print(i, row, "loss =", -log(row["p"][target]))

    w_new = update()
    print("SGD:", W_UL, "→", w_new)
    ids = [0, 1]
    while len(ids) <= len(POSITIONS):
        probabilities = forward(ids, w_ul=w_new)[-1]["p"]
        next_id = max(range(len(TOKENS)), key=probabilities.__getitem__)
        print("Greedy:", repr(TOKENS[next_id]), "p =", probabilities[next_id])
        if next_id == 4:
            break
        ids.append(next_id)
